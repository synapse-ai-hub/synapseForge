from backend.routes.streaming_helpers_v2 import *
from backend.routes.streaming_helpers_v2 import _get_formatted_timestamp, _run_async_search_profile_in_thread, _save_error_feedback_bg
from backend.routes.linkedin_helpers import scrape_profile_from_linkedin
from backend.routes.scraping_linkedin import *
from backend.routes.vnc_manager import start_vnc_for_display


async def conversation_stream(
    message: str,
    conversation_id: Optional[str],
    reasoning_level: str,
    file_contents: List[tuple] = None,
    current_user: dict = None,
    stream_id: Optional[str] = None,
    stream_cancel_event: Optional[Any] = None,  # threading.Event: compartido con scraping_linkedin (run_in_executor, otro hilo)
):
    '''
    Unified function to handle conversations with or without attached files,
    supporting both new and existing conversation flows.

    The function performs the following steps:
    1. Initialization and normalization of optional parameters.
    2. File processing: wraps attached files in XML-like tags.
    3. Conversation management: creates a new conversation or loads an existing one.
    4. Prompt construction: builds the full prompt including context.
    5. LLM streaming: sends the prompt to the LLM and streams chunks to the frontend.
    6. Storage: saves the complete conversation in the ``chat`` collection.
    7. RAG: stores a user-assistant pair in ``user_rag`` with size metadata.

    Args:
        message (str): User message.
        conversation_id (Optional[str]): Existing conversation ID; ``None`` for a new conversation.
        reasoning_level (str): Desired reasoning depth (passed to downstream prompts).
        file_contents (List[tuple], optional): List of ``(filename, text)`` tuples; defaults to ``None``.
        current_user (dict, optional): Authenticated user data containing at least ``user_id`` and ``role``.

    Yields:
        str: Server-Sent Events (SSE) formatted strings containing JSON payloads for the frontend.
    '''



    usage_dict = {
        'prompt_tokens': 0,
        'completion_tokens': 0,
        'total_tokens': 0,
        'total_time': 0
    }
    fecha_actual = datetime.now().strftime('%d/%m/%Y')
    router_result = None

    accumulated_response = ''
    accumulated_reasoning = ''
    validated_output = ''  # Inicializar vacío; se llena si hay herramientas ejecutadas

   
    message_uuid = None
    full_prompt = message
    plan = None
    e_suscription = False
    e_title = False
    e_conversation = False

    def _linkedin_output_str(data):
        """Extrae string de perfiles desde result['data'] (dict con 'output', tuple o 'results')."""
        if data is None:
            return ''
        if isinstance(data, tuple):
            return data[0] if len(data) > 0 else ''
        if isinstance(data, dict):
            # Prioridad 1: output directo (scraping_linkedin o mensaje de no results)
            out = data.get('output')
            if out is not None:
                return out[0] if isinstance(out, (tuple, list)) else (out if isinstance(out, str) else str(out))
            
            # Prioridad 2: results agregados (combination_linkedin)
            results = data.get('results')
            if isinstance(results, list) and len(results) > 0:
                parts = []
                for r in results:
                    if isinstance(r, dict):
                        p_out = r.get('output')
                        if p_out:
                            p_str = p_out[0] if isinstance(p_out, (tuple, list)) else (p_out if isinstance(p_out, str) else str(p_out))
                            if p_str: parts.append(p_str)
                return "\n\n".join(parts)
        return ''

    stream_aborted = False
    _DEBUG_ID = None
    existing_doc = ''
    conversation_data = None
    pipeline_mode = None
    vacante = ''
    try:
        if stream_id and stream_cancel_event is not None:
            yield f"data: {json.dumps({'type': 'stream_id', 'content': stream_id})}\n\n"
        # =====================================================================
        # STEP 1: INITIALIZATION - Normalize optional parameters
        # =====================================================================

        user_id = current_user['user_id']
        role = current_user['role']

        
        if EXECUTION_MODE == 'DEBUG':
            fecha_debug, hora_debug = _get_formatted_timestamp()
            _DEBUG_ID = f'debug_{fecha_debug}_{hora_debug}//{user_id}'
    
            db.add_vector(
                'debug',
                ids=_DEBUG_ID,
                elements=f'# **Debug Session** {fecha_debug} {hora_debug} // {user_id}',
                user_id=user_id,
                role=role,
                metadatas={'user_id': user_id, 'role': role, 'timestamp': f'{fecha_debug}_{hora_debug}'}
            )
            # Enviar external_id al front para E2E (Selenium lo lee de window.__e2e_debug_ids)
            yield f"data: {json.dumps({'type': 'test_debug_id', 'content': _DEBUG_ID})}\n\n"
        
        
        
        # Critical: verify subscription status for regular users
        if role == 'user':
            user_data = db.query_sql('users', where='id = %s', params=(user_id,))
            if user_data:
                user_record = user_data[0]
                expires_at_val = user_record.get('subscription_expires_at')
                is_expired = True
                if expires_at_val:
                    if isinstance(expires_at_val, str):
                        expires_at_dt = datetime.fromisoformat(str(expires_at_val))
                    else:
                        expires_at_dt = expires_at_val
                    if datetime.now() < expires_at_dt:
                        is_expired = False
                if is_expired:
                    response = '⛔ Tu suscripción ha vencido. Para seguir usando el agente, por favor renueva tu plan.'
                    yield f"data: {json.dumps({'type': 'chunk', 'content': response})}\n\n"
                    accumulated_response = response
                    e_suscription = True
                 

        if not e_suscription:
            # Ensure file_contents is a list
            file_contents = file_contents or []

            # Retrieve user name
            nombre_result = db.query_sql('users', 'name', where='id = %s', params=(user_id,))
            nombre = nombre_result[0]['name'] if nombre_result else 'Usuario'

            # Retrieve existing user summary
            row = db.query_sql(
                table="vector_store",
                columns="document",
                where="collection = %s and external_id = %s",
                params=("owner", f'resumen_{user_id}'),
                user_id=user_id,
            )
            resumen = row[0]["document"] if row and len(row) > 0 else 'Sin resumen disponible'

            # Query memory (colección memory, todos los documentos)
            try:
                memory_rows = db.query_sql(
                    table='vector_store',
                    columns='document',
                    where='collection = %s',
                    params=('memory',),
                    user_id=user_id
                )
                if memory_rows:
                    memoria_str = '- ' + '\n- '.join(str(row['document']).strip() for row in memory_rows if row.get('document'))
                else:
                    memoria_str = 'Sin información adicional.'
            except Exception as mem_err:
                memoria_str = 'Sin información adicional.'

            # Load agent config and agent memory
            validation_mode = 'flexible'  # default
            creativity_level = 'default'  # default
            response_style = ''  # free-form user preference
            agent_memory_str = ''  # user custom instructions
            try:
                agent_settings = db.query_sql('user_settings', columns='agent_config, agent_memory', user_id=user_id)
                if agent_settings and len(agent_settings) > 0:
                    settings_row = agent_settings[0]
                    if settings_row.get('agent_config'):
                        agent_config = settings_row['agent_config']
                        validation_mode = agent_config.get('validation_mode', 'flexible')
                        creativity_level = agent_config.get('creativity_level', 'default')
                        response_style = agent_config.get('response_style', '')
                    if settings_row.get('agent_memory'):
                        agent_memory_str = settings_row['agent_memory']
            except Exception as config_err:
                print(f"[WARNING] Error loading agent_config: {config_err}. Using defaults.")

            # Helper function: map creativity_level to (temperature, top_p) overrides for llm_stream calls
            # Returns (None, None) for 'default' to preserve original hardcoded values
            def _get_streaming_params(level: str = 'default') -> tuple:
                """Map creativity_level to temperature and top_p overrides.
                
                Args:
                    level: 'default', 'low', or 'high'
                
                Returns:
                    tuple: (temperature, top_p) or (None, None) for 'default'
                """
                if level == 'low':
                    return (0.6, 0.8)
                elif level == 'high':
                    return (1.5, 0.92)
                else:  # 'default' — no overrides
                    return (None, None)

            # Get temperature and top_p overrides based on creativity level
            temperature_override, top_p_override = _get_streaming_params(creativity_level)

            # =====================================================================
            # STEP 2: FILE PROCESSING
            # =====================================================================
            if file_contents:
                try:
                    MAX_CHARS_PER_FILE = 500_000
                    file_names = ", ".join([fn for fn, _ in file_contents])
                    hidden_section = "<inicio_adjuntos>"
                    parts = []
                    for fn, text in file_contents:
                        if text is None or not isinstance(text, str):
                            text = str(text) if text is not None else "[Sin contenido]"
                        if len(text) > MAX_CHARS_PER_FILE:
                            text = text[:MAX_CHARS_PER_FILE] + f"\n\n[... texto truncado (máx. {MAX_CHARS_PER_FILE} caracteres por archivo) ...]"
                        parts.append(f"**Archivo: {fn}**\n{text}")
                    hidden_section += "\n\n".join(parts)
                    hidden_section += "<fin_adjuntos>"
                    full_prompt = f"{message}\n\n{hidden_section}"
                except Exception as file_section_exc:
                    raise
            else:
                full_prompt = message

            if EXECUTION_MODE == 'DEBUG':
                agent_config_log = "\n".join([
                    "## Configuración del Agente",
                    f"- **Nivel de validación:** {validation_mode}",
                    f"- **Nivel de creatividad:** {creativity_level}",
                    f"- **Estilo de respuesta:** {response_style if response_style else '(no especificado)'}",
                    f"- **Memoria del agente:** {agent_memory_str if agent_memory_str else '(vacía)'}",
                    "",
                    f"## User message:\n\n{full_prompt}"
                ])
                await update_debug(_DEBUG_ID, user_id=user_id, role=role, response=agent_config_log)

            # =====================================================================
            # STEP 3 & 4: CONVERSATION MANAGEMENT & PROMPT CONSTRUCTION
            # =====================================================================
            existing_doc = ''
            conversation_data = None

            if not conversation_id:
                # --- NEW CONVERSATION ---
                rows = db.query_sql(
                    table="vector_store",
                    columns="external_id",
                    where="collection = %s",
                    params=("chat",),
                    user_id=user_id,
                )
                titles_list = [row["external_id"] for row in rows] if rows else []
                if titles_list:
                    titles_formatted = "\n".join([f"{i+1}. {title}" for i, title in enumerate(titles_list)])
                else:
                    titles_formatted = "Sin restricciones (primera conversación)"

                title_prompt = agent.prompt('p_title').format(
                    message=full_prompt,
                    titles=titles_formatted
                )

                title = agent.llm_process(
                    model=agent.model_openai_small,
                    prompt=title_prompt,
                    max_tokens=300,
                    reasoning_format="hidden",
                    reasoning_effort="low"
                )

                if EXECUTION_MODE == 'DEBUG':
                
                    await update_debug(_DEBUG_ID, user_id=user_id, role=role, response=f"## Generated title:\n\n{title}")

                if title['status'] == 'error':
                    response = '❌ Error al guardar la conversación.\nPor favor, intentalo nuevamente más tarde. 🤗\n'
                    print('[DEBUG] Error al generar titulo')
                    accumulated_response = response
                    yield f"data: {json.dumps({'type': 'chunk', 'content': response})}\n\n"
                    e_title = True
                    asyncio.create_task(_save_error_feedback_bg('e_title', message, conversation_id, user_id, message_uuid))

                else:          

                    title = title['data']
                    title = title.strip().replace('"', "").replace("'", "")

                    yield f"data: {json.dumps({'type': 'conversation_id', 'content': title})}\n\n"
                    conversation_id = title
            else:
                if not e_title:
                    # --- EXISTING CONVERSATION ---
                    conversation_data = find_conversation_by_title(conversation_id, user_id)
                    if not conversation_data:
                        message_uuid = str(uuid.uuid4())
                        yield f"data: {json.dumps({'type': 'message_uuid', 'content': message_uuid})}\n\n"
                        yield f"data: {json.dumps({'type': 'conversation_id', 'content': conversation_id})}\n\n"
                        response = 'Conversación no encontrada.'
                        accumulated_response = response
                        yield f"data: {json.dumps({'type': 'chunk', 'content': response})}\n\n"
                        e_conversation = True
                    else:
                        yield f"data: {json.dumps({'type': 'conversation_id', 'content': conversation_id})}\n\n"
                        existing_doc = conversation_data['document']

            if not e_title and not e_conversation:

                # =====================================================================
                # STEP 5: PROMPT CONSTRUCTION & ROUTER
                # =====================================================================

                message_uuid = str(uuid.uuid4())
                yield f"data: {json.dumps({'type': 'message_uuid', 'content': message_uuid})}\n\n"

                if existing_doc:
                    context_prompt = f"{existing_doc}\n\n**Usuario**:\n\n {full_prompt}\n"
                else:
                    context_prompt = f"\n\n**Usuario**:\n\n {full_prompt}\n"

                # Build system_content with agent_memory and response_style injected at the end for Groq prompt caching
                system_content_base = agent.prompt('sc_general').format(
                    nombre=nombre, resumen=resumen, memoria=memoria_str
                )
                
                # Inject agent_memory and response_style at the end (for cache optimization in Groq)
                custom_instructions = []
                if response_style and response_style.strip():
                    custom_instructions.append(f"Estilo de respuesta preferido: {response_style}")
                if agent_memory_str and agent_memory_str.strip():
                    custom_instructions.append(f"Instrucciones del usuario: {agent_memory_str}")
                
                if custom_instructions:
                    system_content = f"{system_content_base}\n\n## Personalizaciones:\n" + "\n".join(custom_instructions)
                else:
                    system_content = system_content_base
                
                corrections_loaded = db.retrieve_vector('corrections', 'CoD_corrections') or ''
                tool_registry = agent.prompt('tool_registry')

                if stream_cancel_event and stream_cancel_event.is_set():
                    yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                    accumulated_response += STREAM_ABORTED_MESSAGE
                    stream_aborted = True

                if not stream_aborted:
                    prompt_router = agent.prompt('p_llm_0_router_v2').format(
                        message=context_prompt, tool_registry=tool_registry
                    )
                    router_result = await agent.router(prompt_router, system_content)

                    if EXECUTION_MODE == 'DEBUG':
                        await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                           response=f"## Router response:\n\n{router_result}")

                    route = (router_result.get('data') or {}).get('route', 'direct')
                    pipeline_mode = (router_result.get('data') or {}).get('pipeline_mode') or None
                    vacante = ''
                    if pipeline_mode == 'recruiting':
                        try:
                            vacante = hidden_section
                        except Exception:
                            vacante = ''

                    if stream_cancel_event and stream_cancel_event.is_set():
                        yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                        accumulated_response += STREAM_ABORTED_MESSAGE
                        stream_aborted = True

                    if not stream_aborted:
                        if route == 'planner':
                            # =====================================================================
                            # PATH: PLANNER (planner → for etapa → dm → validate → execute → validator → stream)
                            # =====================================================================
                            e_credentials = False
                            e_decision = False
                            e_supervisor = False
                            e_tool = False
                            e_validator = False
                            e_inicio_nvc = False
                            e_reject = False
                            e_planner = False
                            validated_output = ''

                            if stream_cancel_event and stream_cancel_event.is_set():
                                yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                accumulated_response += STREAM_ABORTED_MESSAGE
                                stream_aborted = True

                            if not stream_aborted:
                                prompt_planner = agent.prompt('p_llm_1_planner').format(
                                    message=context_prompt,
                                    plans_corrections=corrections_loaded,
                                    fecha=fecha_actual
                                )

                                planner_result = await agent.planner(prompt_planner, system_content)

                                if EXECUTION_MODE == 'DEBUG':
                                    await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                       response=f"## Generated plan\n\n{json.dumps(planner_result, ensure_ascii=False, indent=2)}")

                                if planner_result['status'] in ('canceled', 'error'):
                                    response = '❌ Error al comenzar la búsqueda.\nPor favor, intentalo nuevamente más tarde. 🤗\n'
                                    print('[DEBUG] Error al generar plan')
                                    accumulated_response += response
                                    yield f"data: {json.dumps({'type': 'chunk', 'content': response})}\n\n"
                                    stream_aborted = True
                                    e_planner = True
                                    asyncio.create_task(_save_error_feedback_bg('e_planner', message, conversation_id, user_id, message_uuid))

                            if not stream_aborted and not e_planner:
                                usage_dict = await agent.usage_metrics(usage_dict, **planner_result['usage'])
                                plan = planner_result['data']['plan']
                                planner_pipeline_mode = planner_result['data'].get('pipeline_mode')
                                if planner_pipeline_mode:
                                    pipeline_mode = planner_pipeline_mode
                                if pipeline_mode == 'recruiting':
                                    try:
                                        vacante = hidden_section
                                    except Exception:
                                        vacante = ''

                                reasoning_chunk = 'Pensando un plan en base al mensaje del usuario...\n'
                                accumulated_reasoning += reasoning_chunk
                                yield f"data: {json.dumps({'type': 'reasoning', 'content': reasoning_chunk})}\n\n"
                                await asyncio.sleep(0.1)

                                reasoning_chunk = planner_result['data']['reasoning'] + '\n'
                                accumulated_reasoning += reasoning_chunk
                                yield f"data: {json.dumps({'type': 'reasoning', 'content': reasoning_chunk})}\n\n"
                                await asyncio.sleep(0.1)

                                plan_str = ''
                                for step_item in plan:
                                    plan_str += f"Paso {step_item['step']}: {step_item['description']}\n"
                                accumulated_reasoning += plan_str + '\n'
                                yield f"data: {json.dumps({'type': 'reasoning', 'content': plan_str + '\n'})}\n\n"
                                await asyncio.sleep(0.1)
                                yield f"data: {json.dumps({'type': 'typing_text', 'content': 'executingPlan'})}\n\n"

                                for etapa in plan:
                                    if e_decision or e_supervisor or e_tool or e_credentials or e_validator or e_planner:
                                        break
                                    if stream_cancel_event and stream_cancel_event.is_set():
                                        yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                        accumulated_response += STREAM_ABORTED_MESSAGE
                                        stream_aborted = True
                                        break

                                    description = etapa['description']
                                    tool = etapa['tool']
                                    step = etapa['step']
                                    tool_name = ''
                                    cancel_event = None
                                    vnc_stream_id = None
                                    correcciones_tools = []
                                    count_tools = 1
                                    func = None
                                    args = {}
                                    decision_maker_p = None

                                    # ------ LOOP: decision_maker + validate_tool_call (max 3 intentos) ------
                                    while True:
                                        if stream_cancel_event and stream_cancel_event.is_set():
                                            yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                            accumulated_response += STREAM_ABORTED_MESSAGE
                                            stream_aborted = True
                                            break

                                        corrections_parts = []
                                        for c in correcciones_tools:
                                            if isinstance(c, dict) and c.get('corrections'):
                                                corrections_parts.append(c['corrections'])
                                        corrections_text = '\n\n'.join(corrections_parts) if corrections_parts else ''

                                        try:
                                            tool_examples = agent.prompt(f'tool_{tool}')
                                        except Exception:
                                            tool_examples = f"[No se encontraron ejemplos específicos para la herramienta: {tool}]"

                                        prompt_decision_maker = agent.prompt('p_llm_2_decision_maker').format(
                                            description=description,
                                            tool_name=tool,
                                            tool_registry=tool_registry,
                                            correcciones=corrections_text,
                                            fecha=fecha_actual,
                                            tool_examples=tool_examples,
                                            pipeline_mode=pipeline_mode or '',
                                            vacante=vacante,
                                            validated_output=validated_output if validated_output else 'Ninguna'
                                        )

                                        if stream_cancel_event and stream_cancel_event.is_set():
                                            yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                            accumulated_response += STREAM_ABORTED_MESSAGE
                                            stream_aborted = True
                                            break

                                        decision_maker_p = await agent.decision_maker(prompt_decision_maker)

                                        if EXECUTION_MODE == 'DEBUG':
                                            await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                               response=f"## Decision Maker\nPaso: {step}, Ciclo tool: {count_tools}\n\n{json.dumps(decision_maker_p, ensure_ascii=False, indent=2)}")

                                        if decision_maker_p['status'] in ('canceled', 'error'):
                                            response = '❌ Error al comenzar la búsqueda.\nPor favor, intentalo nuevamente más tarde. 🤗\n'
                                            print('[DEBUG] Error al seleccionar herramientas')
                                            accumulated_response += response
                                            yield f"data: {json.dumps({'type': 'chunk', 'content': response})}\n\n"
                                            e_decision = True
                                            stream_aborted = True
                                            asyncio.create_task(_save_error_feedback_bg('e_decision', message, conversation_id, user_id, message_uuid))
                                            break

                                        usage_dict = await agent.usage_metrics(usage_dict, **(decision_maker_p['usage'] or {}))
                                        tool_name = decision_maker_p['data']['tool_name']

                                        supervisor = validate_tool_call(
                                            decision_maker_p['data']['tool_name'],
                                            decision_maker_p['data'].get('parameters') or {}
                                        )

                                        if EXECUTION_MODE == 'DEBUG':
                                            await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                               response=f"## Tool validation\nPaso: {step}, Ciclo tool: {count_tools}\n\n{json.dumps(supervisor, ensure_ascii=False, indent=2)}")

                                        if supervisor['status'] == 'error':
                                            correcciones_tools.append(supervisor['data'])
                                            count_tools += 1
                                            if count_tools >= 4:
                                                response = '❌ Error al comenzar la búsqueda.\nPor favor, intentalo nuevamente más tarde. 🤗\n'
                                                print('[DEBUG] Error al validar herramienta')
                                                accumulated_response += response
                                                yield f"data: {json.dumps({'type': 'chunk', 'content': response})}\n\n"
                                                e_supervisor = True
                                                stream_aborted = True
                                                asyncio.create_task(_save_error_feedback_bg('e_supervisor', message, conversation_id, user_id, message_uuid))
                                                break
                                            continue

                                        break  # Parámetros válidos

                                    if stream_aborted or e_decision or e_supervisor:
                                        break

                                    # --- Inyección de args ---
                                    args = dict(decision_maker_p['data'].get('parameters') or {})
                                    args['user_id'] = user_id
                                    web_search_original_query = None

                                    if tool_name == 'scraping_linkedin':
                                        args['role'] = role
                                        args['title'] = conversation_id
                                        args['pipeline_mode'] = pipeline_mode
                                        args['_debug_id'] = _DEBUG_ID if EXECUTION_MODE == 'DEBUG' else None
                                        args['vacante'] = vacante
                                        args['user_query'] = full_prompt
                                    elif tool_name == 'rag':
                                        args['user_message'] = message
                                    elif tool_name == 'search_profile':
                                        args['role'] = role
                                        args['_debug_id'] = _DEBUG_ID if EXECUTION_MODE == 'DEBUG' else None
                                        args['pipeline_mode'] = pipeline_mode
                                        args['vacante'] = vacante
                                    elif tool_name == 'special_message':
                                        args['channel'] = args.get('channel', '')
                                        args['tone'] = args.get('tone', '')
                                        args['pipeline_mode'] = pipeline_mode
                                        args['vacante'] = vacante
                                        args['role'] = role
                                        args['_debug_id'] = _DEBUG_ID if EXECUTION_MODE == 'DEBUG' else None
                                        if stream_cancel_event is not None:
                                            args['stream_cancel_event'] = stream_cancel_event
                                    elif tool_name == 'web_search' and 'query' in args:
                                        web_search_original_query = args.get('query', '')

                                    if EXECUTION_MODE == 'DEBUG':
                                        _skip = {'stream_cancel_event', 'cancel_event', 'progress_queue'}
                                        _args_log = {k: v for k, v in args.items() if k not in _skip}
                                        await update_debug(_DEBUG_ID, user_id=user_id, role=role,

                                                asyncio.create_task(_save_error_feedback_bg('e_tool_scraping', message, conversation_id, user_id, message_uuid))

                                        elif tool_name == 'search_profile':
                                            if stream_cancel_event is not None:
                                                args['stream_cancel_event'] = stream_cancel_event
                                            # Phase 1: misma función search_profile (solo DB)
                                            result = await func(**args)
                                            if result.get('status') == 'not_found':
                                                # Phase 2: noVNC + scrape_profile_from_linkedin (no se llama desde el agente)
                                                if is_container:
                                                    vnc_display = db.get_user_vnc_display(user_id)
                                                    if vnc_display is not None:
                                                        if not start_vnc_for_display(vnc_display):
                                                            accumulated_response += FRIENDLY_ERROR_MESSAGE
                                                            yield f"data: {json.dumps({'type': 'chunk', 'content': FRIENDLY_ERROR_MESSAGE})}\n\n"
                                                            stream_aborted = True
                                                            e_tool = True
                                                        if not stream_aborted:
                                                            vnc_url = get_vnc_url_for_user(user_id)
                                                            if vnc_url:
                                                                vnc_stream_id = uuid.uuid4().hex
                                                                cancel_event = threading.Event()
                                                                args['cancel_event'] = cancel_event
                                                                vnc_active_scraping[vnc_stream_id] = {'cancel_event': cancel_event}
                                                                loop = asyncio.get_running_loop()
                                                                vnc_future = loop.create_future()
                                                                vnc_pending_decisions[vnc_stream_id] = vnc_future
                                                                yield f"data: {json.dumps({'type': 'open_vnc', 'content': {'url': vnc_url, 'vnc_stream_id': vnc_stream_id}})}\n\n"
                                                                try:
                                                                    allowed = await asyncio.wait_for(vnc_future, timeout=60.0)
                                                                except asyncio.TimeoutError:
                                                                    allowed = False
                                                                vnc_pending_decisions.pop(vnc_stream_id, None)
                                                                if not allowed:
                                                                    yield f"data: {json.dumps({'type': 'close_vnc'})}\n\n"
                                                                    accumulated_response += FRIENDLY_ERROR_MESSAGE
                                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': FRIENDLY_ERROR_MESSAGE})}\n\n"
                                                                    stream_aborted = True
                                                                    e_tool = True
                                                if not stream_aborted:
                                                    close_vnc_q = queue.Queue()
                                                    args['close_vnc_queue'] = close_vnc_q
                                                    loop = asyncio.get_running_loop()
                                                    search_future = loop.run_in_executor(
                                                        None, _run_async_search_profile_in_thread, scrape_profile_from_linkedin, args
                                                    )
                                                    result = None
                                                    cancel_ev = args.get('cancel_event') if (vnc_stream_id is not None) else None
                                                    search_canceled = False
                                                    try:
                                                        while not search_future.done():
                                                            if stream_cancel_event and stream_cancel_event.is_set():
                                                                result = {'status': 'canceled', 'message': 'Ejecución detenida por el usuario.', 'data': None, 'usage': {}}
                                                                search_canceled = True
                                                                break
                                                            if cancel_ev and getattr(cancel_ev, 'is_set', None) and cancel_ev.is_set():
                                                                result = {'status': 'canceled', 'message': 'Ejecución cancelada por el usuario (ventana noVNC cerrada).', 'data': None, 'usage': {}}
                                                                search_canceled = True
                                                                break
                                                            try:
                                                                if close_vnc_q.get_nowait() == 'close_vnc':
                                                                    yield f"data: {json.dumps({'type': 'close_vnc'})}\n\n"
                                                                    await asyncio.sleep(0.1)
                                                            except queue.Empty:
                                                                pass
                                                            await asyncio.sleep(0.1)

                                                        if not search_canceled and result is None:
                                                            try:
                                                                result = await asyncio.wrap_future(search_future)
                                                            except Exception as e_sp:
                                                                result = {'status': 'error', 'message': str(e_sp), 'data': None, 'usage': {}}
                                                        elif result is None:
                                                            result = {'status': 'canceled', 'message': 'Ejecución detenida por el usuario.', 'data': None, 'usage': {}}

                                                        if result and result.get('status') in ('canceled', 'error'):
                                                            status = result.get('status')
                                                            output_str = _linkedin_output_str(result.get('data'))
                                                            has_data = output_str and 'Ver perfil' in output_str

                                                            if has_data:
                                                                if status == 'error':
                                                                    full_content = f"⚠️ No se pudo completar la búsqueda, pero acá tenés los resultados obtenidos hasta el momento.\n\n{output_str}"
                                                                else: # canceled
                                                                    full_content = output_str
                                                            else: # no data
                                                                if status == 'error':
                                                                    full_content = FRIENDLY_ERROR_MESSAGE.rstrip()
                                                                else: # canceled
                                                                    full_content = 'No se encontraron perfiles.'

                                                            accumulated_response += full_content
                                                            yield f"data: {json.dumps({'type': 'chunk', 'content': full_content})}\n\n"
                                                            stream_aborted = True
                                                            if EXECUTION_MODE == 'DEBUG':
                                                                await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                                                response=f"## Herramienta cancelada/error:\n\n{result.get('message', '')}")
                                                            e_tool = True
                                                            if status == 'error':
                                                                asyncio.create_task(_save_error_feedback_bg('e_tool_search_profile', message, conversation_id, user_id, message_uuid))
                                                    finally:
                                                        if vnc_stream_id is not None:
                                                            vnc_active_scraping.pop(vnc_stream_id, None)
                                                            yield f"data: {json.dumps({'type': 'close_vnc'})}\n\n"
                                                            await asyncio.sleep(0.1)

                                        elif asyncio.iscoroutinefunction(func):
                                            if stream_cancel_event and stream_cancel_event.is_set():
                                                yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                                accumulated_response += STREAM_ABORTED_MESSAGE
                                                stream_aborted = True
                                            else:
                                                # Inyectar query original del usuario para regeneración en herramientas de LinkedIn
                                                if tool_name in ('scraping_linkedin', 'combination_linkedin', 'profile_linkedin'):
                                                    args['user_query'] = message
                                                result = await func(**args)
                                                if result.get('status') in ('canceled', 'error'):
                                                    status = result.get('status')
                                                    output_str = _linkedin_output_str(result.get('data'))
                                                    has_data = output_str and 'Ver perfil' in output_str
                                                    is_linkedin = tool_name in ('scraping_linkedin', 'combination_linkedin', 'profile_linkedin', 'search_profile')

                                                    if is_linkedin:
                                                        if has_data:
                                                            if status == 'error':
                                                                full_content = f"⚠️ No se pudo completar la búsqueda, pero acá tenés los resultados obtenidos hasta el momento.\n\n{output_str}"
                                                            else: # canceled
                                                                full_content = output_str
                                                        else: # no data
                                                            if status == 'error':
                                                                full_content = FRIENDLY_ERROR_MESSAGE.rstrip()
                                                            else: # canceled
                                                                full_content = 'No se encontraron perfiles.'
                                                    else:
                                                        full_content = FRIENDLY_ERROR_MESSAGE.rstrip()

                                                    accumulated_response += full_content
                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': full_content})}\n\n"
                                                    stream_aborted = True
                                                    e_tool = True
                                                    asyncio.create_task(_save_error_feedback_bg('e_tool', message, conversation_id, user_id, message_uuid))

                                        else:
                                            if stream_cancel_event and stream_cancel_event.is_set():
                                                yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                                accumulated_response += STREAM_ABORTED_MESSAGE
                                                stream_aborted = True
                                            else:
                                                if tool_name == 'web_search' and 'query' in args:
                                                    args['query'] = f"{web_search_original_query}\n\nIMPORTANTE: Excluye completamente resultados de redes sociales como Facebook, Instagram, Twitter/X, TikTok, etc. Solo incluye información de fuentes profesionales, noticias, artículos, sitios web oficiales y LinkedIn."
                                                result = func(**args)
                                                if result.get('status') in ('canceled', 'error'):
                                                    status = result.get('status')
                                                    output_str = _linkedin_output_str(result.get('data'))
                                                    has_data = output_str and 'Ver perfil' in output_str
                                                    is_linkedin = tool_name in ('scraping_linkedin', 'combination_linkedin', 'profile_linkedin', 'search_profile')

                                                    if is_linkedin:
                                                        if has_data:
                                                            if status == 'error':
                                                                full_content = f"⚠️ No se pudo completar la búsqueda, pero acá tenés los resultados obtenidos hasta el momento.\n\n{output_str}"
                                                            else: # canceled
                                                                full_content = output_str
                                                        else: # no data
                                                            if status == 'error':
                                                                full_content = FRIENDLY_ERROR_MESSAGE.rstrip()
                                                            else: # canceled
                                                                full_content = 'No se encontraron perfiles.'
                                                    else:
                                                        full_content = FRIENDLY_ERROR_MESSAGE.rstrip()

                                                    accumulated_response += full_content
                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': full_content})}\n\n"
                                                    stream_aborted = True
                                                    e_tool = True
                                                    asyncio.create_task(_save_error_feedback_bg('e_tool', message, conversation_id, user_id, message_uuid))

                                    # Guardar web_search exitoso en DB
                                    if not e_tool and not stream_aborted and tool_name == 'web_search' and result and result.get('status') == 'success':
                                        try:
                                            query_to_save = web_search_original_query if web_search_original_query else message
                                            db.add_sql('web_search', {'conversation_id': conversation_id, 'user_message': query_to_save}, user_id=user_id)
                                        except Exception as web_search_save_error:
                                            print(web_search_save_error)

                                        # Bypass Validador si scraping no devolvió resultados
                                        res_data = result.get('data') or {}
                                        is_zero_results = res_data.get('no_results')
                                        if not e_tool and not stream_aborted and tool_name in ('scraping_linkedin', 'combination_linkedin') and result and result.get('status') == 'success' and is_zero_results:
                                            # Extraer filtros aplicados
                                            f_obj = {
                                                'Ubicación': args.get('locations') or args.get('prospect_locations'),
                                                'Industria': args.get('industries') or args.get('company_industries'),
                                                'Empresas': args.get('companies'),
                                                'Tamaño': args.get('sizes') or args.get('company_sizes')
                                            }
                                            f_str = "\n".join([f"- {k}: {v}" for k, v in f_obj.items() if v]) or "Ningún filtro específico aplicado."

                                            # Extraer empresas y cargos para el reporte
                                            companies_list = res_data.get('found_companies') or []
                                            roles_list = res_data.get('tried_roles') or res_data.get('queries') or []
                                            
                                            # Formatear empresas como lista simple (el LLM generará la tabla)
                                            companies_str = "\n".join([f"- {c}" for c in companies_list]) if companies_list else "No se identificaron empresas válidas para prospectar cargos."
                                            roles_str = "\n".join([f"- {r}" for r in roles_list]) if roles_list else "Búsqueda inicial fallida."

                                            prompt_no_results = agent.prompt('p_llm_no_results').format(
                                                user_query=message,
                                                filters_used=f_str,
                                                found_companies_list=companies_str,
                                                tried_roles_list=roles_str,
                                                last_message=result.get('message', 'No se encontraron resultados.')
                                            )

                                            async for event in agent.llm_streaming(
                                                agent.model_openai_small,
                                                prompt_no_results,
                                                system_content=system_content,
                                                temperature=1.0,
                                                top_p=0.85,
                                                cleaned_output=True,
                                                stream_cancel_event=stream_cancel_event
                                            ):
                                                if event['type'] == 'aborted':
                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                                    accumulated_response += STREAM_ABORTED_MESSAGE
                                                    break
                                                yield f"data: {json.dumps({'type': 'chunk', 'content': event['content']})}\n\n"
                                                accumulated_response += event['content']

                                            stream_aborted = True

                                    # --- Validador de resultado (sin reintentos) ---
                                    if not stream_aborted and not e_tool and result is not None:
                                        if EXECUTION_MODE == 'DEBUG':
                                            if tool_name == 'scraping_linkedin':
                                                _data = result.get('data')
                                                _analysis = (_data.get('analysis', '') if isinstance(_data, dict) else '')
                                                await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                                response=f"## LinkedIn output\n\n### Data:\n\n{_linkedin_output_str(_data)}\n\n### Analysis:\n\n{_analysis}")
                                            else:
                                                await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                                response=f"## Tool output\n\n{json.dumps(result, ensure_ascii=False, indent=2)}")

                                        if stream_cancel_event and stream_cancel_event.is_set():
                                            yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                            accumulated_response += STREAM_ABORTED_MESSAGE
                                            stream_aborted = True
                                        else:
                                            previous_response = (
                                                _linkedin_output_str(result.get('data'))
                                                if tool_name == 'scraping_linkedin'
                                                else (result.get('data') if isinstance(result.get('data'), str) else str(result.get('data', '')))
                                            )

                                            # Skip validator for LinkedIn tools or if validation_mode is 'flexible'
                                            if tool_name in ('scraping_linkedin', 'combination_linkedin', 'profile_linkedin') or validation_mode == 'flexible':
                                                # Bypass validator completely - accumulate result and continue to llm_stream
                                                # (No stream_aborted flag - allows llm_stream execution)
                                                validated_output = f"{previous_response}\n\n"
                                            else:
                                                validator_examples = agent.prompt(f'validator_tool_{tool_name}') or ''
                                                validator_name = f'p_llm_4_validator_{validation_mode}' if validation_mode in ['media', 'estricto'] else 'p_llm_4_validator_v2'
                                                prompt_validator = agent.prompt(validator_name).format(
                                                    previous_response=previous_response,
                                                    user_request=full_prompt,
                                                    fecha=fecha_actual,
                                                    tool_examples=validator_examples
                                                )

                                                validator = await agent.validator(prompt_validator)

                                                if EXECUTION_MODE == 'DEBUG':
                                                    await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                                    response=f"## Validator output\n\n{json.dumps(validator, ensure_ascii=False, indent=2)}")

                                                usage_dict = await agent.usage_metrics(usage_dict, **(validator['usage'] or {}))

                                                if validator['status'] == 'canceled':
                                                    print('[DEBUG] Error al validar respuesta')
                                                    accumulated_response += FRIENDLY_ERROR_MESSAGE
                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': FRIENDLY_ERROR_MESSAGE})}\n\n"
                                                    e_validator = True
                                                    stream_aborted = True
                                                    asyncio.create_task(_save_error_feedback_bg('e_validator', message, conversation_id, user_id, message_uuid))
                                                elif validator['status'] == 'error':
                                                    accumulated_response += FRIENDLY_ERROR_MESSAGE
                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': FRIENDLY_ERROR_MESSAGE})}\n\n"
                                                    e_validator = True
                                                    stream_aborted = True
                                                    asyncio.create_task(_save_error_feedback_bg('e_validator', message, conversation_id, user_id, message_uuid))
                                                else:
                                                    validated_output = f"{validator['data']['output']}\n\n"

                                    # --- LLM final (solo si no hubo error) ---
                                    if not stream_aborted:
                                        if stream_cancel_event and stream_cancel_event.is_set():
                                            yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                            accumulated_response += STREAM_ABORTED_MESSAGE
                                            stream_aborted = True
                                        else:
                                            streaming_prompt = agent.prompt('p_llm_stream').format(
                                                full_prompt=context_prompt,
                                                validated_output=validated_output if validated_output else 'No se recolectó información adicional.'
                                            )
                                            _stream_output_start = len(accumulated_response)
                                            async for event in agent.llm_streaming(
                                                agent.model_openai_small,
                                                streaming_prompt,
                                                system_content=system_content,
                                                temperature=0,
                                                max_tokens=30000,
                                                cleaned_output=True,
                                                stream_cancel_event=stream_cancel_event,
                                                reasoning_format='hidden',
                                                reasoning_effort='medium'
                                            ):
                                                if event['type'] == 'aborted':
                                                    yield f"data: {json.dumps({'type': 'chunk', 'content': STREAM_ABORTED_MESSAGE})}\n\n"
                                                    accumulated_response += STREAM_ABORTED_MESSAGE
                                                    stream_aborted = True
                                                    break
                                                yield f"data: {json.dumps({'type': 'chunk', 'content': event['content']})}\n\n"
                                                accumulated_response += event['content']

                                            if agent.usage and len(agent.usage) >= 3:
                                                total_tokens = agent.usage[2]
                                                if total_tokens > 0:
                                                    db.add_sql(
                                                        'ai_credit',
                                                        {'fecha': datetime.now().date().isoformat(), 'tokens_utilizados': total_tokens},
                                                        user_id=user_id
                                                    )

                                            if EXECUTION_MODE == 'DEBUG':
                                                _stream_final = accumulated_response[_stream_output_start:]
                                                if _stream_final:
                                                    await update_debug(_DEBUG_ID, user_id=user_id, role=role,
                                                                    response=f"## Respuesta final al usuario\n\n{_stream_final}")

        # =====================================================================
        # STEP 6 + [DONE] — siempre al final (goto pattern)
        # =====================================================================
        print(f"[TOKEN_DEBUG] STEP6: message_uuid={message_uuid!r}, conversation_id={conversation_id!r}")
        if message_uuid is not None and conversation_id is not None:
            yield f"data: {json.dumps({'type': 'message_uuid', 'content': message_uuid})}\n\n"
            token_limit_reached, tl_data = await save_step6(
                message_uuid=message_uuid,
                conversation_id=conversation_id,
                full_prompt=full_prompt,
                accumulated_response=accumulated_response,
                accumulated_reasoning=accumulated_reasoning,
                existing_doc=existing_doc,
                conversation_data=conversation_data,
                usage_dict=usage_dict,
                router_result=router_result,
                reasoning_level=reasoning_level,
                user_id=user_id,
                role=role,
                plan=plan,
                debug_id=_DEBUG_ID,
            )
            print(f"[TOKEN_DEBUG] save_step6 ret: token_limit_reached={token_limit_reached}, tl_data={tl_data}")
            if token_limit_reached:
                print(f"[TOKEN_DEBUG] yielding token_limit_reached with {tl_data}")
                yield f"data: {json.dumps({'type': 'token_limit_reached', **tl_data})}\n\n"
            else:
                yield f"data: [DONE]\n\n"
        else:
            print(f"[TOKEN_DEBUG] skip STEP6 (message_uuid or conversation_id None), yielding [DONE]")
            yield f"data: [DONE]\n\n"

    except Exception as e:
        if EXECUTION_MODE == 'DEBUG':
            try:
                _debug_id_err = locals().get('_DEBUG_ID')
                if _debug_id_err:
                    _u_err = locals().get('usage_dict', {'prompt_tokens': 0, 'completion_tokens': 0, 'total_tokens': 0, 'total_time': 0})
                    await update_debug(
                        _debug_id_err,
                        user_id=locals().get('user_id', 'unknown'),
                        role=locals().get('role', 'unknown'),
                        response=f'# Error\n\n{str(e)}\n\n# Usage (Error)\n\n{json.dumps(_u_err, ensure_ascii=False, indent=2)}'
                    )
            except Exception:
                pass

        try:
            asyncio.create_task(
                handle_error(
                    error=e,
                    message=locals().get('message', 'Mensaje no disponible'),
                    conversation_id=locals().get('conversation_id'),
                    user_id=locals().get('user_id', 'unknown'),
                    role=locals().get('role', 'unknown'),
                    message_uuid=locals().get('message_uuid'),
                    existing_doc=locals().get('existing_doc'),
                    stream_aborted=locals().get('stream_aborted', False)
                )
            )
        except Exception as he:
            print(he)

        yield f"data: {json.dumps({'type': 'chunk', 'content': FRIENDLY_ERROR_MESSAGE})}\n\n"
        yield f"data: [DONE]\n\n"
    finally:
        if stream_id:
            stream_cancel_events.pop(stream_id, None)
