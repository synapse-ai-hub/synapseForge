class EmptyQuery(Exception):
    '''
    Exception raised when a required query input is missing or empty.
    '''
    pass

class UserIDException(Exception):
    '''
    Exception raised when a user_id is required but not provided for an operation that
    requires user scoping.
    '''
    pass

class PermissionDeniedException(Exception):
    '''
    Exception raised when a user without sufficient permissions attempts to modify a
    collection that is not user-scoped.
    '''
    pass


class DB:
    '''
    Core database access class providing multi-tenant aware CRUD operations and
    vector store management using PostgreSQL and pgvector.

    This class encapsulates connection handling, user-scoped permission checks,
    and utilities for working with both relational tables and a vector store that
    holds embeddings generated by a SentenceTransformer model.

    ## Attributes

        - DEFAULT_COLLECTIONS (list[str]): List of all collection names managed by the system.
        - USER_FILTERED_COLLECTIONS (set[str]): Subset of collections that require a `user_id` for
          read/write operations.
        - host (str): PostgreSQL host (from environment or default `localhost`).
        - port (int): PostgreSQL port (from environment or default `5432`).
        - user (str): PostgreSQL user (from environment).
        - password (str): PostgreSQL password (from environment).
        - database (str): PostgreSQL database name (from environment).
        - model (SentenceTransformer): Embedding model used to generate vectors.
        - conn (psycopg2.extensions.connection | None): Active database connection.
        - sql (psycopg2.extensions.cursor | None): Cursor for executing SQL statements.

    ## Methods

        - __init__(): Loads environment variables and initializes the SQL connection.
        - init_sql(): Reads connection parameters, creates a SentenceTransformer instance,
          and establishes a PostgreSQL connection.
        - _requires_user_scope(collection_name): Returns True if the collection is in
          `USER_FILTERED_COLLECTIONS`.
        - _validate_write_permissions(collection_name, user_id, role): Checks that a write
          operation respects multi-tenant rules and raises appropriate exceptions.
        - _ensure_user_scope(collection_name, user_id): Ensures a read operation respects
          user scoping, raising `UserIDException` when required.
        - close_connection(): Gracefully closes the cursor and connection.
        - execute_query(query, params=None, fetch_all=True): Executes a raw SQL query and
          returns results as a list of dictionaries.
        - add_user(...): Inserts a new user record and returns the generated UUID.
        - set_user_password(user_id, password_hash): Updates a user's password hash.
        - get_user_by_email(email): Retrieves a user record by email.
        - get_user_by_google_id(google_id): Retrieves a user record by Google ID.
        - update_user_subscription(user_id, status, expires_at=None): Updates subscription fields.
        - update_user_google_tokens(...): Updates Google OAuth tokens with careful handling of
          optional refresh token and expiry.
        - get_user_google_token(user_id): Returns the stored Google access token.
        - get_user_google_refresh_token(user_id): Returns the stored Google refresh token.
        - get_user_google_token_expires_at(user_id): Returns the expiry timestamp of the Google token.
        - get_user_google_token_with_expiry(user_id): Returns both access token and expiry in a dict.
        - add_sql(table, values, user_id=None): Inserts one or many rows into a relational table,
          optionally adding `user_id` for multi-tenant tables.
        - query_sql(...): Executes a SELECT query with optional filtering, ordering, limits,
          and automatic `user_id` scoping.
        - delete_sql(...): Executes a DELETE statement with optional `user_id` scoping.
        - update_sql(...): Executes an UPDATE statement with optional `user_id` scoping.
        - add_vector(...): Inserts documents, metadata, and embeddings into `vector_store`,
          handling user scoping and batch embedding generation.
        - update_vector(...): Updates existing vector_store records, recomputing embeddings only
          when the document changes.
        - get_vector(...): Retrieves ids, documents, and/or metadata from `vector_store` based on
          filters and inclusion flags.
        - delete_vector(...): Deletes records from `vector_store` respecting multi-tenant rules.
        - retrieve_vector(...): Retrieves a single document by collection and external_id.
        - query_vector(...): Performs a semantic similarity search using pgvector, returning ids,
          documents, metadata, and distances.
        - create_embeddings(texts): Generates embeddings for a list of texts using the
          SentenceTransformer model.
    '''

    def __init__(self):
        '''
        Initializes the DB instance by loading environment variables and establishing

        '''
        load_dotenv()
        
        self.init_sql()  
     
    def init_sql(self) -> None:
        '''
        Reads PostgreSQL connection parameters from environment variables, creates a
        SentenceTransformer model for embeddings, and attempts to connect to the
        database. On failure, `self.conn` and `self.sql` are set to `None` and an
        error message is printed.
        '''
        self.host = os.getenv('POSTGRES_HOST', 'localhost')
        self.port = int(os.getenv('POSTGRES_PORT', 5432))
        self.user = os.getenv('POSTGRES_USER')
        self.password = os.getenv('POSTGRES_PASSWORD')
        self.database = os.getenv('POSTGRES_DATABASE')
        self.model = SentenceTransformer('all-mpnet-base-v2')
        
        # Initialize as None to avoid attribute errors if connection fails
        self.conn = None
        self.sql = None

        ssl_mode = os.getenv('POSTGRES_SSL_MODE', 'require')

        try:
            self.conn = psycopg2.connect(
                host=self.host,
                port=self.port,
                user=self.user,
                password=self.password,
                database=self.database,
                cursor_factory=RealDictCursor,
                sslmode=ssl_mode
            )
            self.conn.autocommit = True
            self.sql = self.conn.cursor()
            # print("Conexión a PostgreSQL establecida correctamente.")
                        
        except Exception as e:
            print(f'Error al conectar a PostgreSQL: {str(e)}')
            self.conn = None
            self.sql = None

    def _requires_user_scope(self, collection_name: str) -> bool:
        '''
        Determines whether the given collection requires a `user_id` filter.

        ## Args
            - collection_name (str): Name of the collection to check.

        ## Returns
            - bool: True if the collection is in `USER_FILTERED_COLLECTIONS`, else False.
        '''
        return collection_name in self.USER_FILTERED_COLLECTIONS

    def _validate_write_permissions(self, collection_name: str,
                                    user_id: Optional[str],
                                    role: Optional[str]) -> bool:
        '''
        Validates that a write operation respects multi-tenant permissions.

        ## Args
            - collection_name (str): Target collection.
            - user_id (str | None): Identifier of the user performing the operation.
            - role (str | None): Role of the user (e.g., 'owner').

        ## Returns
            - bool: True if the collection requires user scoping.

        ## Raises
            - UserIDException: If the collection requires a user_id and none is provided.
            - PermissionDeniedException: If a non-owner attempts to modify a collection
              that does not require user scoping.
        '''
        requires_scope = self._requires_user_scope(collection_name)

        if requires_scope and not user_id:
            raise UserIDException(f'Debe proporcionar user_id para la colección {collection_name}')

        if collection_name in self.OWNER_ONLY_COLLECTIONS and role != 'owner':
            raise PermissionDeniedException(
                f'Solo owners pueden modificar la colección {collection_name}'
            )

        # Allow owner, user, and invited roles to modify collections
        if role is not None and role not in ['owner', 'user', 'invited'] and not requires_scope:
            raise PermissionDeniedException(
                f'Solo usuarios con permiso pueden modificar la colección {collection_name}'
            )

        return requires_scope

    def _ensure_user_scope(self, collection_name: str,
                           user_id: Optional[str]) -> bool:
        '''
        Ensures that a read operation includes a `user_id` when required.

        ## Args
            - collection_name (str): Target collection.
            - user_id (str | None): User identifier.

        ## Returns
            - bool: True if the collection requires user scoping.

        ## Raises
            - UserIDException: If the collection requires a user_id and none is provided.
        '''
        requires_scope = self._requires_user_scope(collection_name)

        if requires_scope and not user_id:
            raise UserIDException(f'Debe proporcionar user_id para la colección {collection_name}')

        return requires_scope

    def close_connection(self) -> None:
        '''
        Closes the PostgreSQL cursor and connection, printing a confirmation message.
        '''
        try:
            if self.sql:
                self.sql.close()
            if self.conn:
                self.conn.close()
            print("Conexión a PostgreSQL cerrada correctamente.")
        except Exception as e:
            print(f'Error al cerrar conexión: {str(e)}')

    def execute_query(self, query: str, params: tuple = None, fetch_all: bool = True) -> List[Dict]:
        '''
        Executes a raw SQL query and returns the results as a list of dictionaries.

        ## Args
            - query (str): SQL statement to execute.
            - params (tuple | None): Optional parameters for parameterised queries.
            - fetch_all (bool): If True, fetches all rows; otherwise fetches a single row.

        ## Returns
            - List[Dict]: List of result rows as dictionaries; empty list on error or no results.
        '''
        try:
            if not self.sql:
                print("Error: No hay conexión a la base de datos. Verifica que la conexión se haya establecido correctamente.")
                return []
            
            if params:
                self.sql.execute(query, params)
            else:
                self.sql.execute(query)
            
            if fetch_all:
                results = self.sql.fetchall()
                return [dict(row) for row in results] if results else []
            else:
                result = self.sql.fetchone()
                return [dict(result)] if result else []
        except Exception as e:
            print(f'Error al ejecutar consulta: {str(e)}')
            import traceback
            traceback.print_exc()
            return []

    def add_user(self, email: str, name: str, role: str, google_id: str = None, 
                 subscription_status: str = 'starter', password_hash: str = None,
                 email_verified: bool = None) -> str:
        '''
        Inserts a new user into the `users` table and returns the generated UUID.

        ## Args
            - email (str): User's email address.
            - name (str): Full name of the user.
            - role (str): Role identifier (e.g., 'owner', 'member').
            - google_id (str | None): Optional Google account identifier.
            - subscription_status (str): Subscription tier; defaults to 'starter'.
            - password_hash (str | None): Optional hashed password.
            - email_verified (bool | None): If False, user must verify email; if None, defaults to True.

        ## Returns
            - str | None: UUID of the newly created user, or None on failure.
        '''
        try:
            self.sql.execute("SELECT COALESCE(MAX(vnc_display), 98) + 1 AS next_display FROM users")
            row = self.sql.fetchone()
            vnc_display = int(row['next_display']) if row else 99
            use_verified = email_verified if email_verified is not None else True
            if password_hash:
                self.sql.execute("""
                    INSERT INTO users (email, name, google_id, role, subscription_status, password_hash, vnc_display, email_verified)
                    VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
                    RETURNING id
                """, (email, name, google_id, role, subscription_status, password_hash, vnc_display, use_verified))
            else:
                self.sql.execute("""
                    INSERT INTO users (email, name, google_id, role, subscription_status, vnc_display, email_verified)
                    VALUES (%s, %s, %s, %s, %s, %s, %s)
                    RETURNING id
                """, (email, name, google_id, role, subscription_status, vnc_display, use_verified))
            
            result = self.sql.fetchone()
            user_id = str(result['id'])
            print(f"Usuario {email} agregado con ID: {user_id}")
            return user_id
            
        except Exception as e:
            print(f"Error al agregar usuario: {str(e)}")
            return None

    def update_user_google_tokens(self, user_id: str, access_token: str, refresh_token: str = None, expires_in: int = None) -> bool:
        '''
        Updates Google OAuth tokens for a user, preserving existing refresh token when not provided
        and calculating an expiry timestamp if `expires_in` is supplied.

        ## Args
            - user_id (str): UUID of the user.
            - access_token (str): New Google access token.
            - refresh_token (str | None): New refresh token; if None, the existing token is preserved.
            - expires_in (int | None): Seconds until token expiry; used to compute `google_token_expires_at`.

        ## Returns
            - bool: True if the update affected at least one row, False otherwise.
        '''
        try:
            # Preserve existing refresh token if not supplied
            if not refresh_token:
                existing_refresh = self.get_user_google_refresh_token(user_id)
                if existing_refresh:
                    refresh_token = existing_refresh
                else:
                    pass
            
            # Compute expires_at if expires_in is provided
            expires_at = None
            if expires_in:
                safe_expires_in = max(expires_in - 300, 0)
                expires_at = (datetime.now() + timedelta(seconds=safe_expires_in)).isoformat()
            
            # Build appropriate UPDATE statement based on which optional fields are present
            if refresh_token and expires_at:
                self.sql.execute("""
                    UPDATE users 
                    SET google_access_token = %s, 
                        google_refresh_token = %s, 
                        google_token_expires_at = %s,
                        updated_at = CURRENT_TIMESTAMP
                    WHERE id = %s
                """, (access_token, refresh_token, expires_at, user_id))
            elif refresh_token:
                self.sql.execute("""
                    UPDATE users 
                    SET google_access_token = %s, 
                        google_refresh_token = %s, 
                        updated_at = CURRENT_TIMESTAMP
                    WHERE id = %s
                """, (access_token, refresh_token, user_id))
            elif expires_at:
                self.sql.execute("""
                    UPDATE users 
                    SET google_access_token = %s, 
                        google_token_expires_at = %s,
                        updated_at = CURRENT_TIMESTAMP
                    WHERE id = %s
                """, (access_token, expires_at, user_id))
            else:
                self.sql.execute("""
                    UPDATE users 
                    SET google_access_token = %s, 
                        updated_at = CURRENT_TIMESTAMP
                    WHERE id = %s
                """, (access_token, user_id))
            
            return self.sql.rowcount > 0
        except Exception as e:
            print(f"Error al actualizar tokens de Google: {str(e)}")
            return False
    
    def get_user_google_token(self, user_id: str) -> Optional[str]:
        '''
        Retrieves the stored Google access token for a user.

        ## Args
            - user_id (str): UUID of the user.

        ## Returns
            - str | None: Access token if present and not a placeholder, otherwise None.
        '''
        try:
            self.sql.execute("""
                SELECT google_access_token FROM users WHERE id = %s
            """, (user_id,))
            
            result = self.sql.fetchone()
            if result:
                token = result['google_access_token']
                if token and token != '0' and token != 0:
                    return token
            return None
        except Exception as e:
            print(f"Error al obtener token de Google: {str(e)}")
            return None

        '''

    def get_user_google_token_expires_at(self, user_id: str) -> Optional[str]:
        '''
        Retrieves the expiration timestamp of the Google access token for a user.

        ## Args
            - user_id (str): UUID of the user.

        ## Returns
            - str | None: ISO-formatted timestamp or None if not set.
        '''
        try:
            self.sql.execute("""
                SELECT google_token_expires_at FROM users WHERE id = %s
            """, (user_id,))
            
            result = self.sql.fetchone()
            if result and result['google_token_expires_at']:
                expires_at = result['google_token_expires_at']
                if isinstance(expires_at, datetime):
                    return expires_at.isoformat()
                return str(expires_at)
            return None
        except Exception as e:
            print(f"Error al obtener expires_at del token de Google: {str(e)}")
            return None
    
    def get_user_google_token_with_expiry(self, user_id: str) -> Optional[Dict]:
        '''
        Retrieves both the Google access token and its expiration timestamp in a single query.

        ## Args
            - user_id (str): UUID of the user.

        ## Returns
            - dict | None: Dictionary with keys `access_token` and `expires_at`, or None if not found.
        '''
        try:
            self.sql.execute("""
                SELECT google_access_token, google_token_expires_at 
                FROM users 
                WHERE id = %s
            """, (user_id,))
            
            result = self.sql.fetchone()
            if result and result['google_access_token']:
                access_token = result['google_access_token']
                if access_token and access_token != '0' and access_token != 0:
                    expires_at = result.get('google_token_expires_at')
                    
                    if isinstance(expires_at, datetime):
                        expires_at = expires_at.isoformat()
                    
                    return {
                        'access_token': access_token,
                        'expires_at': str(expires_at) if expires_at else None
                    }
            return None
        except Exception as e:
            print(f"Error al obtener token con expiración: {str(e)}")
            return None

    def add_sql(self, table: str, values: Union[Dict, List[Dict]], 
                user_id: str = None) -> None:
        '''
        Inserts one or many rows into a PostgreSQL table, optionally adding `user_id`
        for collections that support multi-tenant scoping.

        ## Args
            - table (str): Target table name.
            - values (dict | list[dict]): Row(s) to insert.
            - user_id (str | None): User identifier to attach to each row (except `users` table).

        ## Returns
            - None
        '''
        try:
            values_list = [values] if isinstance(values, dict) else values
            if not values_list:
                print(f"No se proporcionaron valores para insertar en {table}.")
                return
            
            # Add user_id automatically if provided and not inserting into users table
            if user_id and table != 'users':
                for value in values_list:
                    value['user_id'] = user_id
            
            cols = list(values_list[0].keys())
            
            # Filter duplicates (maintaining existing logic)
            values_to_insert = []
            for value in values_list:
                if table == 'collector':
                    where_clause = "url = %s"
                    params = (value['url'],)
                    if user_id:
                        where_clause += " AND user_id = %s"
                        params = (value['url'], user_id)
                    
                    self.sql.execute(f"SELECT url FROM collector WHERE {where_clause}", params)
                    if self.sql.fetchone():
                        print(f'{value["url"]} ya existe en {table}. No se insertará.')
                        continue
                elif table == 'prospect' and 'linkedin' in value:
                    # prospect: único por linkedin+user_id (no por nombre, que puede venir vacío del LLM)
                    where_clause = "linkedin = %s"
                    params = (value['linkedin'],)
                    if user_id:
                        where_clause += " AND user_id = %s"
                        params = (value['linkedin'], user_id)
                    
                    self.sql.execute(f"SELECT linkedin FROM {table} WHERE {where_clause}", params)
                    if self.sql.fetchone():
                        print(f'{value.get("linkedin", "")} ya existe en {table}. No se insertará.')
                        continue
                elif table == 'company_search' and 'linkedin' in value:
                    where_clause = "linkedin = %s"
                    params = (value['linkedin'],)
                    if user_id:
                        where_clause += " AND user_id = %s"
                        params = (value['linkedin'], user_id)

                    self.sql.execute(f"SELECT linkedin FROM {table} WHERE {where_clause}", params)
                    if self.sql.fetchone():
                        print(f'{value.get("linkedin", "")} ya existe en {table}. No se insertará.')
                        continue

                elif 'nombre' in value:
                    where_clause = "nombre = %s"
                    params = (value['nombre'],)
                    if user_id:
                        where_clause += " AND user_id = %s"
                        params = (value['nombre'], user_id)
                    
                    self.sql.execute(f"SELECT nombre FROM {table} WHERE {where_clause}", params)
                    if self.sql.fetchone():
                        print(f'{value["nombre"]} ya existe en {table}. No se insertará.')
                        continue
                
                values_to_insert.append(value)
            
            # Insert valid values
            if values_to_insert:
                placeholders = []
                for col in cols:
                    if col == 'fecha':
                        placeholders.append('%s')
                    else:
                        placeholders.append('%s')
                
                placeholders_str = ', '.join(placeholders)
                sql_insert = f"INSERT INTO {table} ({', '.join(cols)}) VALUES ({placeholders_str})"
                
                for value in values_to_insert:
                    self.sql.execute(sql_insert, tuple(value.values()))
                
                print(f"Registros insertados en PostgreSQL. Tabla: {table}")
            else:
                print(f"No hay registros nuevos para insertar en {table}.")
                
        except Exception as e:
            print(f"Error al insertar en PostgreSQL: {str(e)}")

    def query_sql(self, table: str, columns: str = "*", where: str = None, 
                  group_by: str = None, order_by: str = None, desc: bool = True, 
                  nulls_last: bool = False, limit: int = None, params: tuple = None, 
                  user_id: str = None, join: str = None, join_on: str = None) -> List[Dict]:
        '''
        Executes a SELECT query with optional filtering, grouping, ordering, limits,
        and automatic `user_id` scoping for multi-tenant tables.

        ## Args
            - table (str): Table to query.
            - columns (str): Columns to select (default "*").
            - where (str | None): Additional WHERE clause (without the keyword).
            - group_by (str | None): GROUP BY clause.
            - order_by (str | None): ORDER BY clause.
            - desc (bool): If True, adds DESC to ORDER BY.
            - limit (int | None): LIMIT clause.
            - params (tuple | None): Parameters for the WHERE clause.
            - user_id (str | None): User identifier for scoping.
            - join (str | None): Table to LEFT JOIN.
            - join_on (str | None): ON condition for the JOIN.

        ## Returns
            - List[Dict]: List of result rows as dictionaries.
        '''
        try:
            sql_query = f"SELECT {columns} FROM {table}"

            where_conditions = []
            query_params = []

            # Add JOIN before WHERE to avoid ambiguity
            if join:
                if not join_on:
                    raise ValueError("join_on es requerido cuando se especifica join")
                sql_query += f" LEFT JOIN {join} ON {join_on}"

            # Automatic user_id filter if applicable
            if user_id and table not in ['users']:
                self.sql.execute("""
                    SELECT column_name FROM information_schema.columns 
                    WHERE table_name = %s AND column_name = 'user_id'
                """, (table,))

                if self.sql.fetchone():
                    if join:
                        where_conditions.append(f"{table}.user_id = %s")
                    else:
                        where_conditions.append("user_id = %s")
                    query_params.append(user_id)

            if where:
                where_conditions.append(where)
                if params:
                    query_params.extend(params)

            if where_conditions:
                sql_query += f" WHERE {' AND '.join(where_conditions)}"

            if group_by:
                sql_query += f' GROUP BY {group_by}'
            if order_by:
                sql_query += f" ORDER BY {order_by}"
                if desc:
                    sql_query += " DESC"
                if nulls_last:
                    sql_query += " NULLS LAST"
            if limit:
                sql_query += f" LIMIT {limit}"

            self.sql.execute(sql_query, tuple(query_params))
            results = self.sql.fetchall()

            return [dict(row) for row in results]

        except Exception as e:
            print(f'Error en consulta PostgreSQL: {str(e)}')
            return []

    def delete_sql(self, table: str, where: str, params: Union[str, List], 
                   user_id: str = None) -> None:
        '''
        Deletes rows from a PostgreSQL table with optional `user_id` scoping.

        ## Args
            - table (str): Target table.
            - where (str): Column name or condition for the WHERE clause.
            - params (str | list): Value(s) for the WHERE condition.
            - user_id (str | None): User identifier for scoping.

        ## Returns
            - None
        '''
        try:
            where_conditions = []
            query_params = []
            
            if user_id and table not in ['users']:
                where_conditions.append("user_id = %s")
                query_params.append(user_id)
            
            if isinstance(params, (list, tuple)):
                if len(params) == 0:
                    print(f'No se proporcionaron parámetros para borrar en {table}.')
                    return
                if '%s' not in where:  
                    placeholders = ', '.join(['%s'] * len(params))
                    where = f'{where} IN ({placeholders})'
                query_params.extend(params)
            else:
                query_params.append(params)
            
            where_conditions.append(where)
            
            sql_delete = f'DELETE FROM {table} WHERE {" AND ".join(where_conditions)}'
            self.sql.execute(sql_delete, tuple(query_params))

            print(f'Registro(s) eliminado(s) en PostgreSQL. Tabla: {table}, Filas afectadas: {self.sql.rowcount}')

        except Exception as e:
            print(f'Error al borrar en PostgreSQL: {str(e)}')

    def update_sql(self, table: str, values: Dict, where: str, params: tuple, 
                   user_id: str = None) -> None:
        '''
        Updates rows in a PostgreSQL table with optional `user_id` scoping.

        ## Args
            - table (str): Target table.
            - values (dict): Column/value mapping for SET clause.
            - where (str): WHERE condition (e.g., "id = %s").
            - params (tuple): Parameters for the WHERE clause.
            - user_id (str | None): User identifier for scoping.

        ## Returns
            - None
        '''
        try:
            where_conditions = []
            query_params = list(values.values())
            
            if user_id and table not in ['users']:
                where_conditions.append("user_id = %s")
                query_params.append(user_id)
            
            where_conditions.append(where)
            query_params.extend(params)
            
            set_clause = ', '.join([f"{k} = %s" for k in values.keys()])
            sql_update = f"UPDATE {table} SET {set_clause} WHERE {' AND '.join(where_conditions)}"
            
            print(f'[UPDATE_SQL] Tabla: {table}')
            print(f'[UPDATE_SQL] Campos a actualizar: {values}')
            print(f'[UPDATE_SQL] SQL: {sql_update}')
            print(f'[UPDATE_SQL] Parámetros: {query_params}')
            
            self.sql.execute(sql_update, tuple(query_params))
            print(f'Registro(s) actualizado(s) en PostgreSQL. Tabla: {table}, Filas afectadas: {self.sql.rowcount}')

        except Exception as e:
            import traceback
            print(f'[ERROR UPDATE_SQL] Error al actualizar PostgreSQL: {str(e)}')
            print(f'[ERROR UPDATE_SQL] Tabla intentada: {table}')
            print(f'[ERROR UPDATE_SQL] Campos: {values}')
            print(traceback.format_exc())

    def add_vector(self, collection_name: str, ids: Union[str, List[str]],
                   elements: Union[str, List[str], None],
                   metadatas: Union[Dict, List[Dict], None],
                   user_id: Optional[str], role: Optional[str]) -> None:
        '''
        Inserts documents, metadata, and embeddings into the `vector_store` table.

        ## Args
            - collection_name (str): Name of the collection.
            - ids (str | list[str]): External identifiers for the records.
            - elements (str | list[str] | None): Document text(s); None results in a NULL document.
            - metadatas (dict | list[dict] | None): Metadata associated with each document.
            - user_id (str | None): User identifier for multi-tenant scoping.
            - role (str | None): User role; required for permission validation.

        ## Returns
            - None
        '''
        try:
            requires_scope = self._validate_write_permissions(collection_name, user_id, role)

            ids_list = ids if isinstance(ids, list) else [ids]
            ids_list = [str(i) for i in ids_list if i is not None]

            if not ids_list:
                raise EmptyQuery('No se proporcionaron ids para agregar registros en vector_store.')

            total = len(ids_list)

            if isinstance(elements, list):
                elements_list = elements
            elif elements is None:
                elements_list = [None] * total
            else:
                elements_list = [elements]

            if len(elements_list) == 1 and total > 1:
                elements_list = elements_list * total

            if len(elements_list) != total:
                raise ValueError('La cantidad de documentos no coincide con los ids proporcionados.')

            if isinstance(metadatas, list):
                metadatas_list = metadatas
            elif metadatas is None:
                metadatas_list = [None] * total
            else:
                metadatas_list = [metadatas]

            if len(metadatas_list) == 1 and total > 1:
                metadatas_list = metadatas_list * total

            if len(metadatas_list) != total:
                raise ValueError('La cantidad de metadatos no coincide con los ids proporcionados.')

            normalized_metadatas: List[Dict] = []
            documents: List[Optional[str]] = []
            docs_to_embed: List[tuple[int, str]] = []

            for index in range(total):
                raw_metadata = metadatas_list[index]
                if raw_metadata is None:
                    metadata_dict = {}
                elif isinstance(raw_metadata, dict):
                    metadata_dict = dict(raw_metadata)
                else:
                    raise ValueError('Cada metadata debe ser un diccionario.')

                if requires_scope and user_id:
                    metadata_dict.setdefault('user_id', user_id)

                normalized_metadatas.append(metadata_dict)

                raw_document = elements_list[index]
                if raw_document is None:
                    documents.append(None)
                    continue

                document = raw_document if isinstance(raw_document, str) else str(raw_document)
                documents.append(document)
                docs_to_embed.append((index, document))

            embeddings_map: Dict[int, List[float]] = {}
            if docs_to_embed:
                texts = [item[1] for item in docs_to_embed]
                embeddings_array = self.create_embeddings(texts)

                for (position, _), vector in zip(docs_to_embed, embeddings_array):
                    embeddings_map[position] = vector.astype(float).tolist() if isinstance(vector, np.ndarray) else list(vector)

            insert_sql = """
                INSERT INTO vector_store (collection, external_id, user_id, document, metadata, embedding)
                VALUES (%s, %s, %s, %s, %s, %s)
            """

            for idx, external_id in enumerate(ids_list):
                metadata_dict = normalized_metadatas[idx]
                document_value = documents[idx]
                embedding_value = embeddings_map.get(idx)
                row_user_id = user_id if requires_scope else (user_id or metadata_dict.get('user_id'))

                self.sql.execute(
                    insert_sql,
                    (
                        collection_name,
                        external_id,
                        row_user_id,
                        document_value,
                        Json(metadata_dict),
                        embedding_value
                    )
                )

            filter_msg = "con filtrado por usuario" if requires_scope else "sin filtrado"
            print(f"Documentos agregados a vector_store '{collection_name}' {filter_msg}")

        except Exception as e:
            print(f"Error al agregar en vector_store: {str(e)}")


    def update_vector(self, collection_name: str, ids: Union[str, List[str]],
                      user_id: Optional[str], role: Optional[str],
                      elements: Union[str, List[str], None] = None,
                      metadatas: Union[Dict, List[Dict], None] = None) -> None:
        '''
        Updates existing records in `vector_store`, optionally recomputing embeddings
        when the document changes.

        ## Args
            - collection_name (str): Target collection.
            - ids (str | list[str]): External identifiers of records to update.
            - user_id (str | None): User identifier for scoping.
            - role (str | None): User role for permission validation.
            - elements (str | list[str] | None): New document text(s); None leaves unchanged.
            - metadatas (dict | list[dict] | None): New metadata; None leaves unchanged.

        ## Returns
            - None
        '''
        try:
            requires_scope = self._validate_write_permissions(collection_name, user_id, role)

            ids_list = ids if isinstance(ids, list) else [ids]
            ids_list = [str(i) for i in ids_list if i is not None]

            if not ids_list:
                raise EmptyQuery('No se proporcionaron ids para actualizar registros en vector_store.')

            select_sql = """
                SELECT external_id, document, metadata, embedding, user_id
                FROM vector_store
                WHERE collection = %s AND external_id = ANY(%s)
            """
            params: List = [collection_name, ids_list]

            if requires_scope:
                select_sql += " AND user_id = %s"
                params.append(user_id)

            self.sql.execute(select_sql, params)
            rows = self.sql.fetchall()

            if not rows:
                print(f"No se encontraron registros para actualizar en vector_store '{collection_name}'.")
                return

            existing = {row['external_id']: row for row in rows}

            elements_provided = elements is not None
            metadatas_provided = metadatas is not None

            if isinstance(elements, list):
                elements_list = elements
            elif elements is None:
                elements_list = [None] * len(ids_list)
            else:
                elements_list = [elements]

            if len(elements_list) == 1 and len(ids_list) > 1:
                elements_list = elements_list * len(ids_list)

            if len(elements_list) != len(ids_list):
                raise ValueError('La cantidad de documentos no coincide con los ids proporcionados.')

            if isinstance(metadatas, list):
                metadatas_list = metadatas
            elif metadatas is None:
                metadatas_list = [None] * len(ids_list)
            else:
                metadatas_list = [metadatas]

            if len(metadatas_list) == 1 and len(ids_list) > 1:
                metadatas_list = metadatas_list * len(ids_list)

            if len(metadatas_list) != len(ids_list):
                raise ValueError('La cantidad de metadatos no coincide con los ids proporcionados.')

            payloads = []
            docs_to_embed: List[tuple[int, str]] = []

            for idx, external_id in enumerate(ids_list):
                current = existing.get(external_id)
                if not current:
                    print(f"Advertencia: id {external_id} no encontrado en vector_store '{collection_name}'.")
                    continue

                current_doc = current.get('document')
                current_embedding = current.get('embedding')
                current_user_id = current.get('user_id')
                current_metadata = current.get('metadata') or {}

                if elements_provided:
                    candidate = elements_list[idx]
                    if candidate is None:
                        document_value = None
                        embedding_value = None
                        should_embed = False
                    else:
                        document_value = candidate if isinstance(candidate, str) else str(candidate)
                        should_embed = document_value != current_doc
                        embedding_value = None if should_embed else current_embedding
                else:
                    document_value = current_doc
                    embedding_value = current_embedding
                    should_embed = False

                if metadatas_provided:
                    candidate_metadata = metadatas_list[idx]
                    if candidate_metadata is None:
                        metadata_value = {}
                    elif isinstance(candidate_metadata, dict):
                        metadata_value = dict(candidate_metadata)
                    else:
                        raise ValueError('Cada metadata debe ser un diccionario.')
                else:
                    metadata_value = dict(current_metadata) if isinstance(current_metadata, dict) else {}

                if requires_scope and user_id:
                    metadata_value.setdefault('user_id', user_id)

                payload_index = len(payloads)
                payloads.append({
                    'external_id': external_id,
                    'document': document_value,
                    'metadata': metadata_value,
                    'embedding': embedding_value,
                    'user_id': current_user_id
                })

                if should_embed and document_value is not None:
                    docs_to_embed.append((payload_index, document_value))

            if not payloads:
                return

            if docs_to_embed:
                texts = [item[1] for item in docs_to_embed]
                embeddings_array = self.create_embeddings(texts)

                for (position, _), vector in zip(docs_to_embed, embeddings_array):
                    payloads[position]['embedding'] = vector.astype(float).tolist() if isinstance(vector, np.ndarray) else list(vector)

            update_sql = """
                UPDATE vector_store
                SET document = %s,
                    metadata = %s,
                    embedding = %s,
                    updated_at = now()
                WHERE collection = %s AND external_id = %s
            """

            if requires_scope:
                update_sql += " AND user_id = %s"

            for payload in payloads:
                params = [
                    payload['document'],
                    Json(payload['metadata']),
                    payload['embedding'],
                    collection_name,
                    payload['external_id']
                ]

                if requires_scope:
                    params.append(user_id)

                self.sql.execute(update_sql, params)

            filter_msg = "con filtrado por usuario" if requires_scope else "sin filtrado"
            print(f"Documentos actualizados en vector_store '{collection_name}' {filter_msg}")

        except Exception as e:
            print(f"Error al actualizar vector_store: {str(e)}")

    def get_vector(self, collection_name: str,
                   ids: Union[str, List[str], None] = None,
                   where: Optional[Dict] = None,
                   include: Optional[List[str]] = None,
                   limit: Optional[int] = None,
                   where_documents: Optional[Dict[str, str]] = None) -> Dict[str, List]:
        '''
        Retrieves ids, documents, and/or metadata from `vector_store` based on filters.

        ## Args
            - collection_name (str): Target collection.
            - ids (str | list[str] | None): Specific external ids to retrieve.
            - where (dict | None): Metadata filters (key/value pairs).
            - include (list[str] | None): Fields to include; defaults to ids only.
            - limit (int | None): Maximum number of results.
            - where_documents (dict | None): Document text filter with `$contains` key.

        ## Returns
            - dict: Dictionary with keys `ids`, `documents`, `metadatas` (lists).
        '''
        try:
            user_id = None
            metadata_filters: Dict[str, Union[str, List[str]]] = {}

            if where:
                for key, value in where.items():
                    if key == 'user_id':
                        user_id = value
                    else:
                        metadata_filters[key] = value

            requires_scope = self._ensure_user_scope(collection_name, user_id)

            if ids is None:
                ids_list = None
            elif isinstance(ids, list):
                ids_list = [str(i) for i in ids]
            else:
                ids_list = [str(ids)]

            include_set = set(include or [])
            include_set.add('ids')

            where_clauses = ["collection = %s"]
            params: List = [collection_name]

            if ids_list:
                where_clauses.append("external_id = ANY(%s)")
                params.append(ids_list)

            if requires_scope:
                where_clauses.append("user_id = %s")
                params.append(user_id)

            for key, value in metadata_filters.items():
                if isinstance(value, list):
                    where_clauses.append("(metadata ->> %s) = ANY(%s)")
                    params.extend([key, value])
                else:
                    where_clauses.append("(metadata ->> %s) = %s")
                    params.extend([key, value])

            if where_documents:
                contains_value = where_documents.get('$contains')
                if contains_value:
                    where_clauses.append("document ILIKE %s")
                    params.append(f"%{contains_value}%")

            sql_query = f"""
                SELECT external_id, document, metadata
                FROM vector_store
                WHERE {' AND '.join(where_clauses)}
                ORDER BY updated_at DESC
            """

            if limit:
                sql_query += " LIMIT %s"
                params.append(limit)

            self.sql.execute(sql_query, params)
            rows = self.sql.fetchall()

            result = {
                'ids': [row['external_id'] for row in rows],
                'documents': [],
                'metadatas': []
            }

            if 'documents' in include_set:
                result['documents'] = [row.get('document') for row in rows]

            if 'metadatas' in include_set:
                result['metadatas'] = [dict(row.get('metadata') or {}) for row in rows]

            return result

        except Exception as e:
            print(f"Error al obtener registros en vector_store: {str(e)}")
            return {
                'ids': [],
                'documents': [],
                'metadatas': []
            }

    def retrieve_vector(self, collection_name: str, external_id: Union[str, int]) -> str:
        '''
        Retrieves a single document from `vector_store` by collection and external_id.

        ## Args
            - collection_name (str): Target collection.
            - external_id (str | int): Identifier of the document.

        ## Returns
            - str: Document text if found, otherwise an empty string.
        '''
        try:
            self.sql.execute(
                """
                SELECT document
                FROM vector_store
                WHERE collection = %s AND external_id = %s
                ORDER BY updated_at DESC
                LIMIT 1
                """,
                (collection_name, str(external_id))
            )

            result = self.sql.fetchone()
            return result['document'] if result and result.get('document') else ""

        except Exception as e:
            print(f"Error al recuperar documento en vector_store: {str(e)}")
            return ""

    def query_vector(self, collection_name: str,
                     query: Union[str, List[str]],
                     user_id: Optional[str] = None,
                     ids: Union[str, List[str], None] = None,
                     n_results: int = 3,
                     include: Optional[List[str]] = None) -> Dict[str, List]:
        '''
        Performs a semantic similarity search on `vector_store` using pgvector.

        ## Args
            - collection_name (str): Target collection.
            - query (str | list[str]): Query text(s) to embed and search.
            - user_id (str | None): User identifier for scoping; ignored for `rag_system`.
            - ids (str | list[str] | None): Optional filter on external ids.
            - n_results (int): Number of nearest neighbours to return.
            - include (list[str] | None): Fields to include in results; defaults to
              documents, metadatas, and distances.

        ## Returns
            - dict: Dictionary with keys `status`, `message`, `data` (containing ids,
              documents, metadatas, distances), and `usage` (always None).
        '''
        try:
            # Public access for rag_system collection
            if collection_name == 'rag_system':
                requires_scope = False
            else:
                requires_scope = self._ensure_user_scope(collection_name, user_id)

            if query is None:
                raise EmptyQuery('Debe proporcionar query para realizar una consulta semántica.')

            if isinstance(query, list):
                if len(query) == 0:
                    raise EmptyQuery('Debe proporcionar al menos un query.')
                queries = query
            else:
                queries = [query]

            if ids is None:
                ids_list = None
            elif isinstance(ids, list):
                ids_list = [str(i) for i in ids]
            else:
                ids_list = [str(ids)]

            include_set = set(include or ['documents', 'metadatas', 'distances'])

            embeddings_array = self.create_embeddings(queries)

            result = {
                'ids': [],
                'documents': [],
                'metadatas': [],
                'distances': []
            }

            for query_vector, _query in zip(embeddings_array, queries):
                vector_list = query_vector.astype(float).tolist() if isinstance(query_vector, np.ndarray) else list(query_vector)

                where_conditions = ["collection = %s", "embedding IS NOT NULL"]
                params: List = [collection_name]

                if ids_list:
                    where_conditions.append("external_id = ANY(%s)")
                    params.append(ids_list)

                if requires_scope:
                    where_conditions.append("user_id = %s")
                    params.append(user_id)

                # pgvector: <=> = cosine distance (0 = identical, 1 = opposite). ASC = most similar first.
                query_sql = f"""
                    SELECT external_id, document, metadata,
                        embedding <=> %s::vector AS distance
                    FROM vector_store
                    WHERE {' AND '.join(where_conditions)}
                    ORDER BY distance ASC
                    LIMIT %s
                """

                query_params = [vector_list] + params + [n_results]

                # Fuerza sequential scan: IVFFlat busca en toda la tabla y luego filtra;
                # si los candidatos no incluyen la colección, devuelve 0 filas (issue pgvector #301).
                combined_sql = "SET LOCAL enable_indexscan = off;\n" + query_sql
                self.sql.execute(combined_sql, query_params)
                rows = self.sql.fetchall()

                ids_bucket = [row['external_id'] for row in rows]
                documents_bucket = [row.get('document') for row in rows] if 'documents' in include_set else []
                metadatas_bucket = [dict(row.get('metadata') or {}) for row in rows] if 'metadatas' in include_set else []
                distances_bucket = [
                    float(row['distance']) if row.get('distance') is not None else None
                    for row in rows
                ] if 'distances' in include_set else []

                result['ids'].append(ids_bucket)
                result['documents'].append(documents_bucket if 'documents' in include_set else [])
                result['metadatas'].append(metadatas_bucket if 'metadatas' in include_set else [])
                result['distances'].append(distances_bucket if 'distances' in include_set else [])

            return {
                    'status': 'success',
                    'message': 'Búsqueda en RAG completada.',
                    'data': result,
                    'usage': None
                    }

        except Exception as e:
            print(f"Error al consultar vector_store: {str(e)}")
            return {
                    'status': 'error',
                    'message': str(e),
                    'data': {
                                'ids': [],
                                'documents': [],
                                'metadatas': [],
                                'distances': []
                            },
                    'usage': None
                    }

    def rag(self, query: str, user_id: str, user_message: str) -> dict:
        '''
        RAG escalonado: user_rag (20) → LLM filtra (p_rag); si no alcanza, chat (3) → LLM filtra todo.
        Devuelve contrato { status, message, data, usage }.
        '''
        from backend.instances import agent  # Lazy para evitar import circular (db ← instances → db)
        usage_acc = {'prompt_tokens': 0, 'completion_tokens': 0, 'total_tokens': 0, 'total_time': 0.0}

        def _documents_from_result(res: dict) -> List[str]:
            if res.get('status') != 'success' or not res.get('data'):
                return []
            docs = res['data'].get('documents') or []
            if not docs:
                return []
            return docs[0] if isinstance(docs[0], list) else list(docs)

        def _chunks_str(docs: List[str]) -> str:
            return "\n\n".join([f"[{i+1}]\n{doc}" for i, doc in enumerate(docs)])

        def _parse_rag_response(llm_out: dict) -> Optional[dict]:
            raw = (llm_out.get('data') or '').strip()
            if not raw:
                return None
            raw = raw.replace('```json', '').replace('```', '').strip()
            try:
                return json.loads(raw)
            except json.JSONDecodeError:
                return None

        res_ur = self.query_vector('user_rag', query, user_id=user_id, n_results=30, include=['documents'])
        if res_ur.get('status') != 'success':
            return {
                'status': res_ur.get('status', 'error'),
                'message': res_ur.get('message', 'Error en user_rag'),
                'data': None,
                'usage': None,
            }
        docs_ur = _documents_from_result(res_ur)
        if not docs_ur:
            return {
                'status': 'success',
                'message': 'Sin documentos en user_rag.',
                'data': '',
                'usage': usage_acc,
            }

        prompt_template = self.retrieve_vector('prompt', 'p_rag')
        if not prompt_template:
            return {'status': 'error', 'message': 'Prompt p_rag no encontrado.', 'data': None, 'usage': None}
        prompt = prompt_template.format(message=user_message, chunks=_chunks_str(docs_ur))

        out1 = agent.llm_process(agent.model_openai_small, prompt, temperature=0.2, top_p=0.4, max_tokens=10000, reasoning_effort='low')
        for k, v in (out1.get('usage') or {}).items():
            if isinstance(v, (int, float)):
                usage_acc[k] = usage_acc.get(k, 0) + v
        parsed1 = _parse_rag_response(out1)
        if not parsed1:
            return {
                'status': 'error',
                'message': 'No se pudo parsear la respuesta del LLM (p_rag).',
                'data': out1.get('data') or '',
                'usage': usage_acc,
            }
        data1 = parsed1.get('data') or {}
        can_answer = data1.get('can_answer', False)
        relevant_content = data1.get('relevant_content', '') or ''

        if can_answer:
            return {
                'status': 'success',
                'message': parsed1.get('message', 'Filtrado user_rag.'),
                'data': relevant_content,
                'usage': usage_acc,
            }

        res_chat = self.query_vector('chat', query, user_id=user_id, n_results=3, include=['documents'])
        if res_chat.get('status') != 'success':
            return {
                'status': 'success',
                'message': 'user_rag insuficiente; error al consultar chat.',
                'data': relevant_content,
                'usage': usage_acc,
            }
        docs_chat = _documents_from_result(res_chat)
        chunks_second = (relevant_content + '\n\n') if relevant_content else ''
        chunks_second += _chunks_str(docs_chat)
        prompt2 = prompt_template.format(message=user_message, chunks=chunks_second)
        out2 = agent.llm_process(agent.model_openai_small, prompt2, temperature=0.2, top_p=0.4, max_tokens=10000, reasoning_effort='low')
        for k, v in (out2.get('usage') or {}).items():
            if isinstance(v, (int, float)):
                usage_acc[k] = usage_acc.get(k, 0) + v
        parsed2 = _parse_rag_response(out2)
        if not parsed2:
            return {
                'status': 'success',
                'message': 'Filtrado user_rag + chat; fallo parseo segundo LLM.',
                'data': relevant_content,
                'usage': usage_acc,
            }
        data2 = parsed2.get('data') or {}
        final_content = data2.get('relevant_content', '') or relevant_content
        return {
            'status': 'success',
            'message': parsed2.get('message', 'Filtrado user_rag + chat.'),
            'data': final_content,
            'usage': usage_acc,
        }



    def query_context(self, user_id: str) -> dict:
        try:
            owner_result = self.query_sql(
                'vector_store', 
                'document', 
                where="collection = %s AND metadata->>'tipo' IN ('empresa', 'servicio', 'producto')", 
                params=('owner',), 
                user_id=user_id
            )
            
            context = '## Contexto\n\n'
            for i in range(len(owner_result)):
                doc = owner_result[i].get('document', '')
                context += f'### Archivo {i+1}:\n\n{doc}\n\n'
            return {
                    'status': 'success',
                    'message': 'Búsqueda en contexto completada.',
                    'data': context,
                    'usage': None
                    }
        except Exception as e:
            return {
                    'status': 'error',
                    'message': f'Error al buscar contexto.\n{str(e)}',
                    'data': None,
                    'usage': None
                    }
      
            

    def create_embeddings(self, texts: Union[str, List[str]]) -> np.ndarray:
        '''
        Generates embeddings for a single string or a list of strings using the
        SentenceTransformer model.

        ## Args
            - texts (str | list[str]): Text or list of texts to embed.

        ## Returns
            - np.ndarray: 2-D array where each row corresponds to the embedding of a text.
        '''
        if isinstance(texts, str):
            sentences = [texts]
        elif isinstance(texts, list):
            if len(texts) == 0:
                raise EmptyQuery('No se proporcionó texto para generar embeddings.')
            sentences = texts
        else:
            raise ValueError('El parámetro texts debe ser str o List[str].')

        embeddings = self.model.encode(sentences=sentences, convert_to_numpy=True)

        if embeddings.ndim == 1:
            embeddings = np.expand_dims(embeddings, axis=0)

        return embeddings
