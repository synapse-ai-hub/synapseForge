import os
import psycopg2
from dotenv import load_dotenv

def get_env():
    load_dotenv()
    host = os.getenv('POSTGRES_HOST', 'localhost')
    port = int(os.getenv('POSTGRES_PORT', 5432))
    user = os.getenv('POSTGRES_USER')
    password = os.getenv('POSTGRES_PASSWORD')
    database = os.getenv('POSTGRES_DATABASE')
    if not all([host, port, user, password, database]):
        raise RuntimeError("Faltan variables POSTGRES_* en el entorno (.env).")
    return host, port, user, password, database

def connect(dbname, host, port, user, password, autocommit=True):
    conn = psycopg2.connect(
        host=host, port=port, user=user, password=password, database=dbname
    )
    conn.autocommit = autocommit
    return conn

def create_database_if_not_exists(host, port, user, password, database):
    """Equivalente a tu create_db_sql: crea la DB si no existe."""
    with connect('postgres', host, port, user, password) as temp_conn:
        with temp_conn.cursor() as cur:
            cur.execute("SELECT 1 FROM pg_database WHERE datname = %s", (database,))
            if not cur.fetchone():
                cur.execute(f'''CREATE DATABASE "{database}"''')
                print(f'Base de datos {database} creada automáticamente.')
            else:
                print(f'Base de datos {database} ya existe.')

def ensure_extensions(conn):
    """Extensiones requeridas por tu esquema:
       - pgcrypto para gen_random_uuid()
       - vector para VECTOR(768) e índice ivfflat
    """
    with conn.cursor() as cur:
        cur.execute("CREATE EXTENSION IF NOT EXISTS pgcrypto;")
        cur.execute("CREATE EXTENSION IF NOT EXISTS vector;")
    print("Extensiones verificadas (pgcrypto, vector).")

        # debug_metrics: IDs de debug procesados para el dashboard de métricas
        cur.execute("""
            CREATE TABLE IF NOT EXISTS debug_metrics (
                external_id VARCHAR(255) NOT NULL,
                fecha DATE NOT NULL
            );
        """)

        # company_search: resultados de búsquedas de empresas en LinkedIn (sin scoring)
        cur.execute("""
            CREATE TABLE IF NOT EXISTS company_search (
                id SERIAL PRIMARY KEY,
                user_id UUID REFERENCES users(id) ON DELETE CASCADE,
                nombre VARCHAR(255),
                sector VARCHAR(255),
                tamanio VARCHAR(100),
                linkedin VARCHAR(500),
                pipeline_mode VARCHAR(50) DEFAULT 'prospecting',
                fecha DATE
            );
        """)
        cur.execute("CREATE INDEX IF NOT EXISTS idx_company_search_user_id ON company_search(user_id);")
        cur.execute("CREATE INDEX IF NOT EXISTS idx_company_search_linkedin ON company_search(linkedin);")

        # system_updates: track last modification date of system features (e.g. LinkedIn UI fixes)
        cur.execute("""
            CREATE TABLE IF NOT EXISTS system_updates (
                id SERIAL PRIMARY KEY,
                feature_name VARCHAR(255) UNIQUE NOT NULL,
                last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            );
        """)
        cur.execute("CREATE INDEX IF NOT EXISTS idx_system_updates_feature ON system_updates(feature_name);")

    print("Tablas e índices creados/verificados correctamente.")

def main():
    host, port, user, password, database = get_env()
    # 1) Crear la base (si no existe)
    create_database_if_not_exists(host, port, user, password, database)
    # 2) Conectar a la base objetivo y crear extensiones + tablas/índices
    with connect(database, host, port, user, password) as conn:
        ensure_extensions(conn)
        create_tables_and_indexes(conn)

if __name__ == "__main__":
    main()
